cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(TitanAudioLink VERSION 0.1.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Visual Studio 2022 C++ desktop tools on Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(AUDIO_ASAN "Instrument application, SDK and tests with AddressSanitizer" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA "/PDBALTPATH:%_PDB%")
if(AUDIO_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG:FULL)
endif()
add_library(AudioLinkSDK SDK/AudioLink.cpp)
target_link_libraries(AudioLinkSDK PUBLIC ole32 uuid propsys cfgmgr32 avrt)
add_executable(AudioLinkCLI App/Console.cpp)
target_link_libraries(AudioLinkCLI PRIVATE AudioLinkSDK)
add_executable(AudioLinkExample Examples/ListAndTest.cpp)
target_link_libraries(AudioLinkExample PRIVATE AudioLinkSDK)
add_executable(AudioLinkToneExample Examples/GenerateTone.cpp)
target_link_libraries(AudioLinkToneExample PRIVATE AudioLinkSDK)
add_executable(AudioLinkTests Tests/AudioTests.cpp)
target_link_libraries(AudioLinkTests PRIVATE ole32 uuid propsys cfgmgr32 avrt)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/App/Main.cpp")
 add_executable(TitanAudioLink WIN32 App/Main.cpp)
 target_link_libraries(TitanAudioLink PRIVATE AudioLinkSDK user32 gdi32 comctl32 shell32)
 target_link_options(TitanAudioLink PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
 set_target_properties(TitanAudioLink PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
foreach(target AudioLinkCLI AudioLinkExample AudioLinkToneExample AudioLinkTests)
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
enable_testing()
add_test(NAME PcmSignalCorrelationGuards COMMAND AudioLinkTests)
add_test(NAME OfflinePreview COMMAND AudioLinkCLI --preview)
set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Opaque signed TitanAudioLink INF/CAT/SYS directory, optional")
set(AUDIO_PACKAGE_RESOURCE "")
enable_language(RC)
set(AUDIO_LICENSE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/AudioLicenseResource.rc")
file(WRITE "${AUDIO_LICENSE_RESOURCE}" "104 RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/docs/LICENSE-Microsoft-Sample.txt\"\n105 RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/THIRD-PARTY-NOTICES.md\"\n")
if(DRIVER_PACKAGE_DIRECTORY)
 file(TO_CMAKE_PATH "${DRIVER_PACKAGE_DIRECTORY}" audio_package)
 foreach(name TitanAudioLink.inf TitanAudioLink.cat TitanAudioLink.sys)
  if(NOT EXISTS "${audio_package}/${name}")
   message(FATAL_ERROR "Driver package is incomplete: ${name}")
  endif()
 endforeach()
 set(AUDIO_PACKAGE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/AudioDriverPackage.rc")
 file(WRITE "${AUDIO_PACKAGE_RESOURCE}" "101 RCDATA \"${audio_package}/TitanAudioLink.inf\"\n102 RCDATA \"${audio_package}/TitanAudioLink.cat\"\n103 RCDATA \"${audio_package}/TitanAudioLink.sys\"\n")
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Installer/Setup.cpp")
 add_library(AudioLinkSetup STATIC Installer/Setup.cpp)
 target_include_directories(AudioLinkSetup PUBLIC Installer)
 target_link_libraries(AudioLinkSetup PUBLIC setupapi newdev cfgmgr32 wintrust crypt32 advapi32 ole32)
 add_executable(TitanAudioLinkSetup WIN32 Installer/Main.cpp ${AUDIO_PACKAGE_RESOURCE} ${AUDIO_LICENSE_RESOURCE})
 add_executable(TitanAudioLinkSetupConsole Installer/Console.cpp ${AUDIO_PACKAGE_RESOURCE} ${AUDIO_LICENSE_RESOURCE})
 foreach(target TitanAudioLinkSetup TitanAudioLinkSetupConsole)
  target_link_libraries(${target} PRIVATE AudioLinkSetup user32 gdi32 comctl32)
  target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
  if(NOT AUDIO_ASAN)
   target_link_options(${target} PRIVATE /DEPENDENTLOADFLAG:0x800)
  endif()
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 endforeach()
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/InstallerTests.cpp")
  add_executable(AudioLinkInstallerTests Tests/InstallerTests.cpp ${AUDIO_LICENSE_RESOURCE})
  target_link_libraries(AudioLinkInstallerTests PRIVATE AudioLinkSetup)
  set_target_properties(AudioLinkInstallerTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  add_test(NAME InstallerIdentityAndSignatureGuards COMMAND AudioLinkInstallerTests)
 endif()
endif()
